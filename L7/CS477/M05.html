void decl_resolve( struct decl *d ) {
    if(!d) return;
    symbol_t kind = scope_level() > 1 ? SYMBOL_LOCAL : SYMBOL_GLOBAL;
    d->symbol = symbol_create(kind, d->type, d->name);
    expr_resolve(d->value);
    scope_bind(d->name, d->symbol);
    if(d->code) {
        scope_enter();
        param_list_resolve(d->type->params);
        stmt_resolve(d->code);
        scope_exit();
    }
    decl_resolve(d->next);
}

boolean type_equals( struct type *a, struct type *b ) {
    if( a->kind == b->kind ) {
        if( a and b are atomic types ) return true;
        else if ( both are array ) return true if subtype is recursively equal;
        else if ( both are function ) return true if both subtype and params are recursively equal;
    }
    return false;
}
