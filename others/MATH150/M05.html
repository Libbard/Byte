procedure binary search(x: integer, a1, a2, ..., an: increasing integers)
i := 1
j := n

  m := floor((i + j)/2)

procedure insertion sort(a1, ..., an: real numbers with n >= 2)
for j := 2 to n
  i := 1
  while aj > ai
    i := i + 1
  m := aj
  for k := 0 to j - i - 1
    aj-k := aj-k-1
  ai := m
