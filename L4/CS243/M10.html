procedure insertion(T: binary search tree, x: item)
v := root of T
while v ≠ null and label(v) ≠ x
  if x < label(v) then
    if left child of v ≠ null then v := left child of v
    else add new vertex as left child of v and set v := null
  else
    if right child of v ≠ null then v := right child of v
    else add new vertex as right child of v and set v := null
if root of T = null then add vertex v to tree and label with x
else if v is null or label(v) ≠ x then label new vertex with x
