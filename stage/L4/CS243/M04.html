$$n! = n \cdot (n-1)!, \text{ base: } 0! = 1$$
Pseudocode
procedure factorial(n: nonnegative integer)
if n = 0 then return 1
else return n * factorial(n - 1)
$$O(n \log n)$$
Pseudocode
procedure mergesort(L = a_1, a_2, ..., a_n)
if n > 1 then
  m := floor(n/2)
  L_1 := a_1, ..., a_m
  L_2 := a_{m+1}, ..., a_n
  L := merge(mergesort(L_1), mergesort(L_2))

NextCombination(a_1 a_2 ... a_r)

while (a[i] > n - r + i)
  i = i - 1
a[i] = a[i] + 1
for (j = i + 1; j <= r; j++)
  a[j] = a[i] + j - i
