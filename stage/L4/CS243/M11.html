procedure inorder(T: ordered rooted tree)
r := root of T
if r is a leaf then list r
else
    l := first child of r from left to right
    T(l) := subtree with l as its root
    inorder(T(l))
    list r
    for each child c of r except for l from left to right
        T(c) := subtree with c as its root
        inorder(T(c))
Pseudocode
procedure postorder(T: ordered rooted tree)
r := root of T
for each child c of r from left to right
    T(c) := subtree with c as its root
    postorder(T(c))

$$\min \sum_{e \in T} weight(e)$$
Pseudocode
procedure Prim (G: weighted connected undirected graph with n vertices)
T := a minimum-weight edge
for i := 1 to n − 2
    e := an edge of minimum weight incident to a vertex in T and not forming a simple circuit in T if added to T
    T := T with e added
return T
Pseudocode
procedure Kruskal (G: weighted connected undirected graph with n vertices)
T := empty graph
for i := 1 to n − 1
    e := any edge in G with the smallest weight that does not form a simple circuit when added to T
    T := T with e added
return T
