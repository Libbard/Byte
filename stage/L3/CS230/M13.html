public static long factorial(long number) {
    if (number <= 1) // test for base case
        return 1; // base cases: 0! = 1 and 1! = 1
    else // recursion step
        return number * factorial(number - 1);
}

// Iterative Factorial
public long factorial(long number) {
    long result = 1;
    for (long i = number; i >= 1; i--) {
        result *= i;
    }
    return result;
}
