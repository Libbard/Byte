public class Scope {
    private static int x = 1; // Class field

    public static void main(String[] args) {
        int x = 5; // Local variable shadows field x
        System.out.println("local x in main is " + x);
        useField();
    }

    public static void useField() {
        System.out.println("field x is " + x); // Accesses class field
    }
}
java 
class Adder {
    // Overloading by parameter types
    static int add(int a, int b) {
        return a + b;
    }
    static double add(double a, double b) {
        return a + b;
    }
    // Overloading by number of parameters
    static int add(int a, int b, int c) {
        return a + b + c;
    }
}

class OverloadingCalculation1 {
    static void sum(int a, long b) {
        System.out.println("int, long method invoked");
    }
    static void sum(int a, int b, int c) {
        System.out.println("int, int, int method invoked");
    }
    public static void main(String args[]) {
        sum(20, 20); // second int literal promoted to long
        sum(20, 20, 20);
    }
}
