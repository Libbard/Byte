class Superclass {
    int i = 20;
}
class Subclass extends Superclass {
    int i = 100;
    void display() {
        System.out.println("i value = " + i); // Prints 100
        System.out.println("superclass i value = " + super.i); // Prints 20
    }
}
java 
class Animal { void eat() { System.out.println("Animals Eat"); } }
class herbivores extends Animal { void eat() { System.out.println("Herbivores Eat Plants"); } }

class Main {
    public static void main(String args[]) {
        Animal a = new Animal();
        Animal h = new herbivores(); // Upcasting
        a.eat(); // Prints: Animals Eat
        h.eat(); // Prints: Herbivores Eat Plants
    }
}

abstract class Animal {
    public abstract void animalSound(); // Abstract method (no body)
    public void sleep() { // Regular method
        System.out.println("Zzz");
    }
}

class Pig extends Animal {
    public void animalSound() {
        System.out.println("The pig says: wee wee");
    }
}

java 
interface Animal {
    public void animalSound();
    public void sleep();
}

class Pig implements Animal {
    public void animalSound() {
        System.out.println("The pig says: wee wee");
    }
    public void sleep() {
        System.out.println("Zzz");
    }
}
